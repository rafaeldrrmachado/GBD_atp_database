use atp;

db.atpplayers.aggregate([
  // 1. Agrupar por 'LinkPlayer', identificador único
  {
    $group: {
      _id: "$LinkPlayer", // O LinkPlayer é a Chave Primária

      // Pega o primeiro valor encontrado para este LinkPlayer
      PlayerName: { $first: "$PlayerName" },
      Born: { $first: "$Born" },
      Height: { $first: "$Height" },
      Hand: { $first: "$Hand" }
    }
  },

  // 2. Reformatar a saída
  {
    $project: {
      _id: 0, 
      LinkPlayer: "$_id", 
      PlayerName: 1,
      Born: 1,
      Height: 1,
      Hand: 1
    }
  },

  // 3. Escreve os docs numa colecao nova
  {
    $out: "players_collection"
  }
]);

db.atpplayers.aggregate([
    // 1. Agrupar pelo campo 'Oponent' para encontrar todos os nomes únicos
    {
        $group: {
            _id: "$Oponent"
        }
    },
    
    // 2. Reformatar a saída
    {
        $project: {
            _id: 0,
            PlayerName: "$_id"
        }
    },
    
    // 3. Escreve os docs numa colecao nova
    {
        $out: "oponents_collection" 
    }
]);

db.atpplayers.aggregate([
    // 1. Agrupar por "Evento" (os campos que definem um torneio único)
    {
        $group: {
            _id: {
                TournamentName: "$Tournament",
                Date: "$Date",
                Location: "$Location"
            },
            
            // Pega no primeiro valor de 'Ground' e 'Prize' encontrados neste grupo.
            Ground: { $first: "$Ground" },
            Prize: { $first: "$Prize" }
        }
    },
    
    // 2. Reformatar a saída
    {
        $project: {
            _id: 0,
            TournamentName: "$_id.TournamentName",
            Date: "$_id.Date",
            Location: "$_id.Location",
            Ground: "$Ground",
            Prize: "$Prize"
        }
    },
    
    // 3. Escreve os docs numa colecao nova
    { 
        $out: "tournaments_collection" // O nome da sua nova coleção de torneios
    }
]);


db.atpplayers.aggregate([
  // 1. Cria a chave normalizada de jogadores, mas mantem os campos originiais
  {
    $project: {
      // Docs do ficheiro
      PlayerName: 1,
      Born: 1,
      Height: 1,
      Hand: 1,
      LinkPlayer: 1,
      Tournament: 1,
      Location: 1,
      Date: 1,
      Ground: 1,
      Prize: 1,
      GameRound: 1,
      GameRank: 1,
      Oponent: 1,
      WL: 1,
      Score: 1,

      // Chave normalizada
      p1: { $min: ["$PlayerName", "$Oponent"] },
      p2: { $max: ["$PlayerName", "$Oponent"] }
    }
  },

  // 2. Ordena para dar prioridade aos Ws
  {
    $sort: {
      WL: -1
    }
  },

  // 3. Agrupa pela chave normalizada
  {
    $group: {
      _id: {
        // Chave única por partida
        tour: "$Tournament",
        loc: "$Location",
        date: "$Date",
        round: "$GameRound",
        p1: "$p1",
        p2: "$p2"
      },
      // $first escolhe preferenciamente o W
      uniqueDoc: { $first: "$$ROOT" }
    }
  },

  // 4. Limpa a estrutura para manter apenas dados do doc
  {
    $replaceRoot: { newRoot: "$uniqueDoc" }
  },

  // 5. Remove 'p1' e 'p2'
  {
    $project: {
      p1: 0,
      p2: 0
    }
  },

  // 6. Escreve os docs numa colecao nova
  {
    $out: "unique_matches_collection"
  }
]);
